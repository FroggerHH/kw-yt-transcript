```bash
uv sync                                            # окружение по uv.lock
uv run --locked --with pytest -- python -m pytest -q -p no:cacheprovider
uvx ruff@0.14.5 check kw_yt_transcript tests       # линтер E/F/W/I/UP
python -m kw_yt_transcript --help
```
